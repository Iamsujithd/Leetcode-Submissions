// Title: Immediate Food Delivery II
            // Difficulty: Medium
            // Language: MySQL
            // Link: https://leetcode.com/problems/immediate-food-delivery-ii/

select 
 as immediate_percentage 
 from Delivery d 
 join (
    select customer_id,min(order_date) as mind
    from Delivery 
    group by customer_id
 ) as b
 on d.customer_id=b.customer_id 
 and d.order_date=b.mind
 round(
    sum(case when b.mind=d.customer_pref_delivery_date then 1 else 0 end)
    *100/count(distinct d.customer_id),2)
